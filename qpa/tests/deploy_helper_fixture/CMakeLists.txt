if(TEST_DEPLOY_QPA)
    if(TEST_STALE_QPA_METADATA)
        # Model an unrelated installed SDK that left valid-looking QPA metadata in the parent scope.
        # A local/source RemoteAccess acquisition must not consume this plugin when its own source
        # configure did not create HyRemote::QpaPlatform.
        set(HyRemote_QPA_AVAILABLE TRUE)
        set(_fixture_qpa_payload
            "${CMAKE_CURRENT_BINARY_DIR}/${CMAKE_SHARED_MODULE_PREFIX}stale-qhyremote${CMAKE_SHARED_MODULE_SUFFIX}")
        file(WRITE "${_fixture_qpa_payload}" "fixture")
        set(HyRemote_QPA_PLUGIN_FILE "${_fixture_qpa_payload}")
    elseif(TEST_QPA_AVAILABLE)
        set(HyRemote_QPA_AVAILABLE TRUE)
        if(TEST_INSTALLED_PAYLOAD)
            # Mirror the real qhyremote MODULE filename exactly: Linux carries the `lib` module
            # prefix while Windows does not. A platform-inaccurate fixture could hide package metadata regressions.
            set(_fixture_qpa_payload
                "${CMAKE_CURRENT_BINARY_DIR}/${CMAKE_SHARED_MODULE_PREFIX}qhyremote${CMAKE_SHARED_MODULE_SUFFIX}")
            file(WRITE "${_fixture_qpa_payload}" "fixture")
            set(HyRemote_QPA_PLUGIN_FILE "${_fixture_qpa_payload}")
        else()
            add_library(qpa-probe MODULE EXCLUDE_FROM_ALL dummy.cpp)
            add_library(HyRemote::QpaPlatform ALIAS qpa-probe)
            if(TEST_STALE_QPA_NEGATIVE_METADATA)
                # Current source target exists and must remain authoritative even when unrelated
                # installed-package metadata says QPA is unavailable / qualified for another Qt.
                set(HyRemote_QPA_AVAILABLE FALSE)
            endif()
        endif()
    else()
        set(HyRemote_QPA_AVAILABLE FALSE)
    endif()
else()
    set(HyRemote_QPA_AVAILABLE FALSE)
endif()
if(TEST_STALE_QPA_NEGATIVE_METADATA)
    set(HyRemote_QPA_QT_VERSION "6.8.2")
else()
    set(HyRemote_QPA_QT_VERSION "6.8.3")
endif()

if(TEST_INSTALLED_PAYLOAD)
    set(_fixture_remoteaccess "${CMAKE_CURRENT_BINARY_DIR}/${CMAKE_SHARED_LIBRARY_PREFIX}HyRemoteRemoteAccess${CMAKE_SHARED_LIBRARY_SUFFIX}")
    file(WRITE "${_fixture_remoteaccess}" "fixture")
    add_library(HyRemote::RemoteAccess SHARED IMPORTED GLOBAL)
    set_target_properties(HyRemote::RemoteAccess PROPERTIES IMPORTED_LOCATION "${_fixture_remoteaccess}")
else()
    add_library(fake-remoteaccess SHARED EXCLUDE_FROM_ALL dummy.cpp)
    add_library(HyRemote::RemoteAccess ALIAS fake-remoteaccess)
endif()

# Source availability is proven by the current-configure hyremote-qml target; installed availability
# is represented only by the existing import-root metadata from HyRemoteConfig.cmake. The stale case
# deliberately provides a plausible absolute path without a source module target.
if(TEST_QML)
    if(TEST_QML_AVAILABLE)
        set(HyRemote_QML_IMPORT_PATH "${CMAKE_CURRENT_BINARY_DIR}/qml-import")
        if(TEST_QML_IMPORT_PATH_EXISTS)
            file(MAKE_DIRECTORY "${HyRemote_QML_IMPORT_PATH}")
        endif()
        if(NOT TEST_INSTALLED_PAYLOAD)
            add_library(qml-backing SHARED EXCLUDE_FROM_ALL dummy.cpp)
            add_library(hyremote-qml ALIAS qml-backing)
            add_library(qml-plugin MODULE EXCLUDE_FROM_ALL dummy.cpp)
            set_property(GLOBAL PROPERTY HYREMOTE_QML_SOURCE_DEPLOY_TARGETS
                "qml-backing;qml-plugin")
        endif()
    elseif(TEST_STALE_QML_METADATA)
        set(HyRemote_QML_IMPORT_PATH "${CMAKE_CURRENT_BINARY_DIR}/stale-qml-import")
        file(MAKE_DIRECTORY "${HyRemote_QML_IMPORT_PATH}")
    endif()
endif()

function(qt_generate_deploy_app_script)
    set(options NO_UNSUPPORTED_PLATFORM_ERROR)
    set(oneValueArgs TARGET OUTPUT_SCRIPT)
    cmake_parse_arguments(ARG "${options}" "${oneValueArgs}" "" ${ARGN})
    if(TEST_QML)
        message(FATAL_ERROR "ordinary Qt deploy command called for QML probe")
    endif()
    if(NOT ARG_TARGET STREQUAL "deploy-probe")
        message(FATAL_ERROR "unexpected deployment target: ${ARG_TARGET}")
    endif()
    if(NOT ARG_NO_UNSUPPORTED_PLATFORM_ERROR)
        message(FATAL_ERROR "HyRemote must preserve NO_UNSUPPORTED_PLATFORM_ERROR")
    endif()
    set(script "${CMAKE_CURRENT_BINARY_DIR}/qt-app-deploy.cmake")
    file(WRITE "${script}" "# ordinary deployment probe\n")
    set(${ARG_OUTPUT_SCRIPT} "${script}" PARENT_SCOPE)
    set(APP_DEPLOY_CALLED TRUE PARENT_SCOPE)
endfunction()

function(qt_generate_deploy_qml_app_script)
    set(options NO_UNSUPPORTED_PLATFORM_ERROR DEPLOY_USER_QML_MODULES_ON_UNSUPPORTED_PLATFORM)
    set(oneValueArgs TARGET OUTPUT_SCRIPT)
    cmake_parse_arguments(ARG "${options}" "${oneValueArgs}" "" ${ARGN})
    if(NOT TEST_QML)
        message(FATAL_ERROR "QML Qt deploy command called for ordinary QPA probe")
    endif()
    if(NOT ARG_TARGET STREQUAL "deploy-probe")
        message(FATAL_ERROR "unexpected deployment target: ${ARG_TARGET}")
    endif()
    if(NOT ARG_NO_UNSUPPORTED_PLATFORM_ERROR OR
       NOT ARG_DEPLOY_USER_QML_MODULES_ON_UNSUPPORTED_PLATFORM)
        message(FATAL_ERROR "HyRemote QML deployment options regressed")
    endif()
    get_target_property(_probe_qml_imports "${ARG_TARGET}" QT_QML_IMPORT_PATH)
    if(NOT _probe_qml_imports OR NOT HyRemote_QML_IMPORT_PATH IN_LIST _probe_qml_imports)
        message(FATAL_ERROR
            "HyRemote QML import root was not attached before Qt deploy generation: ${_probe_qml_imports}")
    endif()
    set(script "${CMAKE_CURRENT_BINARY_DIR}/qt-qml-deploy.cmake")
    file(WRITE "${script}" "# QML deployment probe\n")
    set(${ARG_OUTPUT_SCRIPT} "${script}" PARENT_SCOPE)
    set(QML_DEPLOY_CALLED TRUE PARENT_SCOPE)
endfunction()

include("${HYREMOTE_SOURCE_DIR}/cmake/HyRemoteDeploy.cmake")

if(TEST_QML)
    if(TEST_DEPLOY_QPA)
        hyremote_deploy(TARGET deploy-probe QML QPA)
    else()
        hyremote_deploy(TARGET deploy-probe QML)
    endif()
    if(NOT QML_DEPLOY_CALLED OR APP_DEPLOY_CALLED)
        message(FATAL_ERROR "QML deployment must preserve exclusive QML high-level dispatch")
    endif()
else()
    hyremote_deploy(TARGET deploy-probe QPA)
    if(NOT APP_DEPLOY_CALLED OR QML_DEPLOY_CALLED)
        message(FATAL_ERROR "ordinary QPA must preserve ordinary high-level deployment dispatch")
    endif()
endif()

get_target_property(_manual_dependencies deploy-probe MANUALLY_ADDED_DEPENDENCIES)
if(NOT _manual_dependencies)
    set(_manual_dependencies "")
endif()

if(TEST_INSTALLED_PAYLOAD)
    if(_manual_dependencies)
        message(FATAL_ERROR
            "installed payloads must not become local build dependencies: ${_manual_dependencies}")
    endif()
else()
    set(_expected_dependencies fake-remoteaccess)
    if(TEST_DEPLOY_QPA AND TEST_QPA_AVAILABLE AND NOT TEST_STALE_QPA_METADATA)
        list(APPEND _expected_dependencies qpa-probe)
    endif()
    if(TEST_QML AND TEST_QML_AVAILABLE)
        list(APPEND _expected_dependencies qml-backing qml-plugin)
    endif()
    foreach(_expected IN LISTS _expected_dependencies)
        if(NOT _expected IN_LIST _manual_dependencies)
            message(FATAL_ERROR
                "source deploy payload was not pulled into application build: ${_expected}; got ${_manual_dependencies}")
        endif()
    endforeach()
    list(LENGTH _manual_dependencies _dependency_count)
    list(LENGTH _expected_dependencies _expected_count)
    if(NOT _dependency_count EQUAL _expected_count)
        message(FATAL_ERROR
            "unexpected source deploy build dependency set: ${_manual_dependencies}")
    endif()
endif()
