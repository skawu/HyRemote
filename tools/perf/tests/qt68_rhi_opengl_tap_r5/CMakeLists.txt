cmake_minimum_required(VERSION 3.21)
project(HyRemoteQt68RhiOpenGlTapR5 LANGUAGES CXX)
find_package(Qt6 6.8.3 REQUIRED COMPONENTS Core Gui Quick)
add_executable(hyremote-qt68-rhi-opengl-r5 main.cpp)
target_compile_features(hyremote-qt68-rhi-opengl-r5 PRIVATE cxx_std_17)
target_link_libraries(hyremote-qt68-rhi-opengl-r5 PRIVATE Qt6::Core Qt6::Gui Qt6::Quick)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(hyremote-qt68-rhi-opengl-r5 PRIVATE -Wall -Wextra -Werror)
endif()
